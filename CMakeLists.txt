cmake_minimum_required(VERSION 3.25)
project(jmcomic VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto SSL)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FTXUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FTXUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FTXUI_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(FTXUI_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install OFF CACHE BOOL "" FORCE)
foreach(dep ftxui json stb curl libwebp)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/${dep}/.git")
    message(FATAL_ERROR "Missing ${dep}: run git submodule update --init --recursive")
  endif()
endforeach()
add_subdirectory(external/ftxui EXCLUDE_FROM_ALL)
add_subdirectory(external/json EXCLUDE_FROM_ALL)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(CURL_USE_OPENSSL ON CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
set(CURL_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
set(CURL_ENABLE_EXPORT_TARGET OFF CACHE BOOL "" FORCE)
set(HTTP_ONLY ON CACHE BOOL "" FORCE)
# Prevent curl from adding its own test suite to this application's CTest run.
set(_jm_tests ${BUILD_TESTING})
set(BUILD_TESTING OFF)
add_subdirectory(external/curl EXCLUDE_FROM_ALL)
set(BUILD_TESTING ${_jm_tests})
foreach(tool ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS FUZZTEST)
  set(WEBP_BUILD_${tool} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(external/libwebp EXCLUDE_FROM_ALL)
add_library(jmcore
  src/models.cpp src/storage.cpp src/auth/crypto.cpp src/api/http.cpp
  src/api/client.cpp src/image/image.cpp src/downloader/manager.cpp)
target_include_directories(jmcore PUBLIC include PRIVATE external/stb external/libwebp/src)
target_link_libraries(jmcore PUBLIC nlohmann_json::nlohmann_json Threads::Threads
  OpenSSL::Crypto PRIVATE CURL::libcurl_static webpdemux webpdecoder)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(jmcore PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  set_source_files_properties(src/image/image.cpp PROPERTIES COMPILE_OPTIONS "-Wno-conversion;-Wno-missing-field-initializers")
endif()
add_executable(jmcomic src/main.cpp src/tui/app.cpp)
target_link_libraries(jmcomic PRIVATE jmcore ftxui::component ftxui::dom ftxui::screen)
install(TARGETS jmcomic RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(jmcomic_tests tests/core_tests.cpp)
  target_include_directories(jmcomic_tests PRIVATE external/stb external/libwebp/src)
  target_link_libraries(jmcomic_tests PRIVATE jmcore webp)
  add_test(NAME core COMMAND jmcomic_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:jmcomic_tests>)
    set_tests_properties(integration PROPERTIES TIMEOUT 60)
    if(UNIX)
      add_test(NAME tui_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tui_smoke.py $<TARGET_FILE:jmcomic>)
      set_tests_properties(tui_smoke PROPERTIES TIMEOUT 35)
    endif()
  endif()
endif()
